```python
async def mofy_user(self, email: str, update_data: UserSchema,
                    session: Annotated[AsyncSession, Depends(get_session)]):
    user_exist = await self.get_user_by_email(email, session)
    if user_exist:
        for field, value in update_data.model_dump(exclude_unset=True).items():
            setattr(user_exist, field, value)
```

> 在之前fastapi学习中我使用的一般是自己写死一个接收到的参数然后再做update_data.model_dump()
> 譬如修改用户时候要修改id,修改昵称,但是uid肯定是不变的,那么之前写的就得专门把uid拿出来，例如下面这个伪代码
```python
async def mofy_user_old_version(self,email:str,
                                update_data:UserSchema,
                                session:Annotated[AsyncSession,Depends(get_session)]):
    user_exist=await self.get_user_by_email(email,session)
    if user_exist:
        update_data.uid=user_exist.uid 
        session.execute(update(User),[user_exist,model_dump(exclude_unset=True)])
        session.commit()
```
